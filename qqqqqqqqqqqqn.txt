@pytest.mark.django_db(transaction=True)
def test_csv_import_is_idempotent(auth_client, user):
    
    csv_content = b"type,amount,category,date,description\n"                 #  Готовим CSV-файл в памяти
    csv_content += b"expense,150.50,Food,2026-11-01,Lunch\n"
    csv_content += b"income,1000.00,Salary,2026-11-05,November\n"
    # SimpleUploadedFile — это способ имитировать файл, который пользователь загружает через форму. Нам не нужен реальный файл на диске, достаточно «байтов в памяти».      
    csv_file = SimpleUploadedFile(
        "test.csv", 
        csv_content, 
        content_type="text/csv"
    )
    response1 = auth_client.post('/upload-csv/', {'file': csv_file})                    #  Загружаем файл ПЕРВЫЙ.  проверяет, что импорт работает
    assert response1.status_code == 302  
    file_name = default_storage.save("test.csv", csv_file)
    _ = import_csv_task(user.id, file_name)                                                                           # Редирект после успеха
    assert Transaction.objects.count() == 2                                           # Создалось 2 транзакции      

    csv_file_2 = SimpleUploadedFile(
        "test.csv", 
        csv_content, 
        content_type="text/csv"
    )
    response2 = auth_client.post('/upload-csv/', {'file': csv_file_2})
    print(f"Количество транзакций после первого импорта: {Transaction.objects.count()}")
    print(f"Response context: {response1.context}") 
    assert response2.status_code == 302
    # Проверяем, что количество НЕ изменилось (дубликаты не создались)
    assert Transaction.objects.count() == 2  # это контракт. Мы говорим: «Система гарантирует, что повторный импорт не создает дублей».

from django.contrib import messages
from django.contrib.auth.decorators import login_required
from django.core.files.storage import default_storage
from django.shortcuts import redirect, render

from .forms import CSVUploadForm, TransactionForm
from .tasks import import_csv_task


@login_required
def add_transaction(request):
    if request.method == 'POST':
        form = TransactionForm(request.POST, user=request.user)                # Передаем user
        if form.is_valid():
            transaction = form.save(commit=False)                              # form.save(commit=False) — мы не сохраняем сразу, потому что нам нужно добавить owner (владельца). Без commit=False Django попытался бы сохранить транзакцию без владельца и выдал бы ошибку.
            transaction.owner = request.user                                # Привязываем владельца
            transaction.save()                                           # Сохраняем
            return redirect('transaction_list')                            # после успешного сохранения перенаправляем на список транзакций (чтобы не было повторной отправки формы при обновлении страницы).
    else:
        form = TransactionForm(user=request.user)

    return render(request, 'tracker/add_transaction.html', {'form': form})

@login_required
def upload_csv(request):
    if request.method == 'POST':
        form = CSVUploadForm(request.POST, request.FILES)

        if form.is_valid():
            csv_file = request.FILES['file']  

            if not csv_file.name.endswith('.csv'):
                messages.error(request, 'Файл должен быть в формате CSV.')
                return redirect('upload_csv')   
                     
            file_name = default_storage.save(csv_file.name, csv_file)        #  сохраняет файл на диск (например, в /app/media/), чтобы Celery мог его прочитать из другого процесса.                    
            import_csv_task.delay(request.user.id, file_name)       # Запускаем задачу в фоне Метод .delay() не выполняет задачу сразу. Он кладет задачу в очередь Redis и сразу возвращает управление. Пользователь сразу видит сообщение «Файл принят в обработку», не дожидаясь, пока парсинг закончится.    
            messages.success(request, 'Файл принят в обработку. Результат появится чуть позже.')  # Воркер Celery (отдельный процесс) забирает задачу из Redis и выполняет import_csv_task.
            return redirect('transaction_list')
        
    else:
        form = CSVUploadForm()

    return render(request, 'tracker/upload_csv.html', {'form': form})

@pytest.mark.django_db
def test_user_cannot_modify_other_users_transaction(auth_client, user):
    """
    Проверяет, что пользователь не может получить, изменить или удалить
    чужую транзакцию. Все три операции должны возвращать 404.
    """
    # Создаем второго пользователя и его транзакцию
    User = get_user_model()
    user2 = User.objects.create_user(username='user2', password='password2')
    cat2 = Category.objects.create(name="Food", owner=user2)
    t2 = Transaction.objects.create(
        type='income', 
        amount=100, 
        category=cat2, 
        owner=user2, 
        date='2026-11-02'
    )
    
    # Логинимся под ПЕРВЫМ пользователем
    auth_client.force_authenticate(user=user)
    
    # Пытаемся получить чужую транзакцию
    response_get = auth_client.get(f'/api/transactions/{t2.id}/')
    assert response_get.status_code == 404
    
    # Пытаемся изменить чужую транзакцию
    response_patch = auth_client.patch(f'/api/transactions/{t2.id}/', {'amount': '999.99'})
    assert response_patch.status_code == 404
    
    # Пытаемся удалить чужую транзакцию
    response_delete = auth_client.delete(f'/api/transactions/{t2.id}/')
    assert response_delete.status_code == 404
    
    # Проверяем, что транзакция не изменилась и не удалилась
    t2.refresh_from_db()
    assert t2.amount == 100
    assert Transaction.objects.filter(id=t2.id).exists()

from django.conf import settings
from django.db import models
from django.utils import timezone


class Category(models.Model):
    name = models.CharField(max_length=100, verbose_name="Name") # verbose_name — это название для админки.
    owner = models.ForeignKey(
        settings.AUTH_USER_MODEL,                  # ссылка на кастомного пользователя
        on_delete=models.CASCADE,                  # если пользователя удалят, все его категории тоже удалятся.
        related_name='categories',                 # позволяет из пользователя достать его категории: user.categories.all().
        verbose_name="Owner"                       # verbose_name — это название для админки (человеческое).
    )                                              # Поле owner — связь «многие к одному» (FK). Один пользователь может иметь много категорий.

    class Meta:                                    # Названия таблицы, поля фильтрации, индексы. Hастройки модели, которые не создают колонок в БД.
        verbose_name = "Category"                  # как модель называется в админке (в единственном и множественном числе).
        verbose_name_plural = "Categories"         # как модель называется в админке (в единственном и множественном числе).
        constraints = [                                                                         # Это шов на уровне БД. UniqueConstraint запрещает дубликаты.
            models.UniqueConstraint(fields=['owner', 'name'], name='unique_category_per_user')  # Но не просто «уникальное имя», а уникальная пара (owner + name). 
        ]                                                                                       # То есть у одного пользователя не может быть двух категорий «Food», но у разных пользователей — может.

    def __str__(self):
        return self.name


class Transaction(models.Model):              
    TYPE_CHOICES = [                            # чтобы ограничить возможные значения (только income или expense).
        ('income', 'Income'),                   # Константа TYPE_CHOICES — список кортежей. Она нужна для поля type, 
        ('expense', 'Expense'),
    ]
    amount = models.DecimalField(
        max_digits=12, 
        decimal_places=2, 
        verbose_name="Amount"
    )                                                        # Шов 2: Decimal для денег (не float!)
    type = models.CharField(                                 # Поле type — строка. Но благодаря choices=TYPE_CHOICES, 
        max_length=10,                                       # Django будет проверять, что значение входит в список (income/expense). 
        choices=TYPE_CHOICES,                                # В БД хранится строка, а в админке отображается человеческое название.
        verbose_name="Type"
    )
    category = models.ForeignKey(
        Category, 
        on_delete=models.PROTECT,                          # Это защита. Если попытаться удалить категорию, в которой есть транзакции, база данных запретит это (выбросит ProtectedError). Мы не хотим терять финансовые данные.
        related_name='transactions',                       # из категории можно достать её транзакции: category.transactions.all().
        verbose_name="Category"
    )                                                      # Шов 3: Категория как FK, а не строка
    owner = models.ForeignKey(
        settings.AUTH_USER_MODEL, 
        on_delete=models.CASCADE, 
        related_name='transactions',
        verbose_name="Owner"
    )                                                                                      # Шов 4: owner через настройку, а не auth.User. Поле owner — FK на пользователя.
    date = models.DateField(default=timezone.now, verbose_name="Date", db_index=True)
    description = models.CharField(max_length=255, blank=True, verbose_name="Description") # description — описание. blank=True означает, что поле необязательное (может быть пустым).
    created_at = models.DateTimeField(auto_now_add=True)

    class Meta:                                                   #Названия таблицы, поля фильтрации, индексы.
        verbose_name = "Transaction"                              # Название таблицы
        verbose_name_plural = "Transactions"                      # Название таблицы
        ordering = ['-date', '-created_at']
        indexes = [
        models.Index(fields=['owner', 'date']),                    
        ]                                                         # По умолчанию: сначала новые
        constraints = [                                            #  Защита от дублей при импорте
            models.UniqueConstraint(
                fields=['owner', 'date', 'amount', 'type', 'description'],
                name='unique_transaction_per_user'
            )
        ]
    def __str__(self):
        return f"{self.get_type_display()} {self.amount} ({self.date})"

import csv
from datetime import datetime
from io import TextIOWrapper

from celery import shared_task
from django.contrib.auth import get_user_model
from django.core.files.storage import default_storage
from django.db import IntegrityError

from .models import Category, Transaction

User = get_user_model()

@shared_task                                        #  @shared_task — декоратор, который говорит Celery, что это задача.
def import_csv_task(user_id, file_path):
    try:
        user = User.objects.get(id=user_id)                  # default_storage.open() — открывает файл, который Django сохранил на диск (в папку MEDIA_ROOT). Это работает, если ты настроил MEDIA_ROOT в settings.py.
        with default_storage.open(file_path, 'rb') as f:     # Открываем файл из хранилища (Django сохранил его на диск)
            data_set = TextIOWrapper(f, encoding='utf-8')
            reader = csv.DictReader(data_set)            
            created_count = 0
            skipped_count = 0                               # <--- Считаем пропущенные

            for row in reader:
                category_name = row.get('category', '').strip()
                category, _ = Category.objects.get_or_create(owner=user, name=category_name)
                
                try:
                    date_obj = datetime.strptime(row['date'], '%Y-%m-%d').date()
                except (ValueError, KeyError):
                    date_obj = datetime.now().date()

                try:
                    Transaction.objects.create(
                    owner=user,
                    type=row['type'],
                    amount=row['amount'],
                    category=category,
                    date=date_obj,
                    description=row.get('description', '')
                )
                    created_count += 1
                except IntegrityError:                   
                    skipped_count += 1        # Если сработал UniqueConstraint, значит, транзакция уже есть
                    continue # <--- Переходим к следующей строке
                        
            default_storage.delete(file_path) # # Удаляем файл после обработки, чтобы не засорять диск
            return f'Успешно загружено: {created_count}. Пропущено дублей: {skipped_count}.'
            
    except (ValueError, KeyError, IntegrityError) as e:
        return f'Ошибка: {e}'

    client_max_body_size 10M;

# finance-tracker
Finance Tracker — это веб-приложение для учета личных доходов и расходов. Проект позволяет добавлять транзакции вручную, импортировать выписки из банка в формате CSV (с защитой от дубликатов) и просматривать историю операций.

🔗 Живая демонстрация: [http://194.113.35.11/](http://194.113.35.11/)

🚀 Стек технологий
- Backend: Python 3.12, Django 5.2, Django REST Framework

- База данных: PostgreSQL

- Асинхронность: Celery, Redis (для фоновой обработки CSV)

- Инфраструктура: Docker, Docker Compose, Nginx, Gunicorn

- Тестирование: Pytest, Pytest-Django

- CI/CD: GitHub Actions (линтер Ruff, тесты)

⚙️ Основные возможности

- Аутентификация: Кастомная модель пользователя (CustomUser), доступ к данным только для авторизованных пользователей.

- CRUD транзакций: Создание, просмотр, фильтрация доходов и расходов.

- Импорт CSV: Загрузка файлов с транзакциями. Задача обрабатывается в фоне через Celery. Реализована идемпотентность (повторная загрузка того же файла не создает дубликатов).

- Админ-панель: Управление категориями и транзакциями.
🛠 Запуск локально (через Docker)

Для запуска вам понадобятся Docker и Docker Compose.

Клонируйте репозиторий:
```bash
git clone https://github.com/mr-sim3003/finance-tracker.git
cd finance-tracker
```
Создайте файл `.env` в корне проекта на основе `.env.example`:
```bash
   cp .env.example .env
```
(Убедитесь, что в .env указаны корректные SECRET_KEY, DATABASE_URL и ALLOWED_HOSTS).

Для запуска тестов без Docker активируйте venv
Запустите контейнеры:
```bash
docker compose up -d --build
```
Примените миграции:
```bash
docker compose exec web python manage.py migrate
```
Создайте суперпользователя (для доступа в админку):
```bash
docker compose exec web python manage.py createsuperuser
```
Приложение будет доступно по адресу: [http://localhost/](http://localhost/)

🧪 Тестирование

Проект покрыт тестами (API, валидация, идемпотентность импорта).

Запуск тестов внутри контейнера:
```bash
docker compose exec web pytest
```
без контейнера.
```bash
pytest
```
